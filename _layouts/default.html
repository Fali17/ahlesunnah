// --- Integrated Client-Side Universal Search & Highlighter Engine --- const searchTriggerBtn = document.getElementById('searchTriggerBtn'); const searchOverlayPanel = document.getElementById('searchOverlayPanel'); const closeSearchBtn = document.getElementById('closeSearchBtn'); const customSearchInput = document.getElementById('customSearchInput'); const clearSearchBtn = document.getElementById('clearSearchBtn'); const dropdownSearchResults = document.getElementById('dropdownSearchResults'); let searchDb = []; const isPostPage = window.location.pathname.includes('/202') || !!document.querySelector('.post-content, article'); // Load database layout maps fetch("{{ '/search.json' | relative_url }}") .then(res => res.json()).then(data => searchDb = data).catch(err => console.error(err)); // Open & Close Menu Triggers searchTriggerBtn.addEventListener('click', () => { searchOverlayPanel.classList.add('is-active'); setTimeout(() => customSearchInput.focus(), 150); }); function closeSearchPanel() { searchOverlayPanel.classList.remove('is-active'); customSearchInput.value = ''; clearSearchBtn.style.display = 'none'; dropdownSearchResults.innerHTML = ''; if (isPostPage) removeHighlights(); } closeSearchBtn.addEventListener('click', closeSearchPanel); // Clear text box mechanic trigger clearSearchBtn.addEventListener('click', () => { customSearchInput.value = ''; clearSearchBtn.style.display = 'none'; dropdownSearchResults.innerHTML = ''; customSearchInput.focus(); if (isPostPage) removeHighlights(); }); // Dynamic Query Routing Evaluators customSearchInput.addEventListener('input', function() { const query = this.value.toLowerCase().trim(); clearSearchBtn.style.display = query.length > 0 ? 'block' : 'none'; dropdownSearchResults.innerHTML = ''; if (query.length < 2) { if (isPostPage) removeHighlights(); return; } // CONTEXT A: If on a single Post Page -> Perform real-time live keyword text highlighting if (isPostPage) { highlightTextOnPostPage(query); } // CONTEXT B: Generate matching post suggestions dropdown (Essential on main index page) const matches = searchDb.filter(item => item.title.toLowerCase().includes(query) || item.excerpt.toLowerCase().includes(query)); if (matches.length > 0) { matches.forEach(item => { const link = document.createElement('a'); link.href = item.url; link.className = 'drop-res-item'; link.innerHTML = `
${item.title}
${item.excerpt}
`; dropdownSearchResults.appendChild(link); }); } else if (!isPostPage) { dropdownSearchResults.innerHTML = '
No matches found
'; } }); // Real-time DOM Text Highlight Node Injection Loops function highlightTextOnPostPage(keyword) { removeHighlights(); const targetArea = document.querySelector('.post-content') || document.querySelector('article') || document.querySelector('.site-main'); if (!targetArea) return; const walk = document.createTreeWalker(targetArea, NodeFilter.SHOW_TEXT, null, false); const textNodes = []; while (walk.nextNode()) textNodes.push(walk.currentNode); textNodes.forEach(node => { const parent = node.parentNode; if (parent.tagName === 'SCRIPT' || parent.tagName === 'STYLE' || parent.classList.contains('search-highlight')) return; const text = node.nodeValue; const index = text.toLowerCase().indexOf(keyword); if (index >= 0) { const span = document.createElement('span'); span.className = 'search-highlight-container'; const matchText = text.substring(index, index + keyword.length); const beforeText = text.substring(0, index); const afterText = text.substring(index + keyword.length); span.innerHTML = `${escapeHtml(beforeText)}${escapeHtml(matchText)}${escapeHtml(afterText)}`; parent.replaceChild(span, node); } }); } function removeHighlights() { document.querySelectorAll('.search-highlight-container').forEach(span => { const parent = span.parentNode; parent.replaceChild(document.createTextNode(span.textContent), span); parent.normalize(); }); } function escapeHtml(str) { return str.replace(/&/g, "&").replace(//g, ">"); }
